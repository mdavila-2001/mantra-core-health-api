// Arnés de prueba (fuera del repo): emite el SQL del nivel 1-2 de TerminologySeedService
// (fuente, code system, versión y TODOS los conceptos internos) con los mismos ids.
import { CONCEPTS, CONCEPT_DEFS, SEED, deterministicId } from '/home/user/mantra-core-health-api/src/common/constants/concepts';
import { MODULE_CONCEPT_SEEDS } from '/home/user/mantra-core-health-api/src/common/seed/module-concepts';
const q = (s: string) => `'${s.replace(/'/g, "''")}'`;
const out: string[] = [];
out.push(`INSERT INTO terminology.terminology_sources (id, code, name, created_at, updated_at) VALUES (${q(SEED.sourceId)}, ${q(SEED.sourceCode)}, 'Mantra Core Internal Terminology', now(), now()) ON CONFLICT DO NOTHING;`);
out.push(`INSERT INTO terminology.code_systems (id, source_id, internal_code, name, canonical_url, case_sensitive, created_at, updated_at) VALUES (${q(SEED.codeSystemId)}, ${q(SEED.sourceId)}, ${q(SEED.codeSystemInternalCode)}, 'Mantra Core Internal Code System', ${q(SEED.canonicalUrl)}, true, now(), now()) ON CONFLICT DO NOTHING;`);
out.push(`INSERT INTO terminology.code_system_versions (id, code_system_id, version, published_at, is_default, created_at, updated_at) VALUES (${q(SEED.codeSystemVersionId)}, ${q(SEED.codeSystemId)}, ${q(SEED.version)}, now(), true, now(), now()) ON CONFLICT DO NOTHING;`);
const catalog = new Map<string, { code: string; display: string }>();
for (const name of Object.keys(CONCEPT_DEFS)) catalog.set((CONCEPTS as Record<string, string>)[name], { code: (CONCEPT_DEFS as any)[name].code, display: (CONCEPT_DEFS as any)[name].display });
for (const seed of MODULE_CONCEPT_SEEDS) catalog.set(deterministicId(seed.key), { code: seed.key, display: seed.display });
for (const [id, c] of catalog) out.push(`INSERT INTO terminology.catalog_concepts (id, code_system_version_id, code, display, created_at, updated_at) VALUES (${q(id)}, ${q(SEED.codeSystemVersionId)}, ${q(c.code)}, ${q(c.display)}, now(), now()) ON CONFLICT DO NOTHING;`);
console.log(out.join('\n'));
