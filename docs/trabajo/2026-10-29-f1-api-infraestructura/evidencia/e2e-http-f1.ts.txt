import { execFileSync } from 'node:child_process';

import { CONCEPTS } from '/home/user/mantra-core-health-api/src/common/constants/concepts';

const API = 'http://localhost:3100';
const env = Object.fromEntries(
  readFileSync('/home/user/mantra-core-health-api/.env', 'utf8')
    .split('\n')
    .filter((l) => /^[A-Z_]+=/.test(l))
    .map((l) => [l.slice(0, l.indexOf('=')), l.slice(l.indexOf('=') + 1).trim()]),
);
const sql = (q: string): string =>
  execFileSync('psql', ['-h', 'localhost', '-p', '5433', '-U', 'mantra', '-d', 'alovida', '-At', '-c', q], {
    env: { ...process.env, PGPASSWORD: env.DB_PASSWORD },
  })
    .toString()
    .trim();

let pass = 0;
let fail = 0;
function check(name: string, ok: boolean, info?: unknown) {
  if (ok) pass += 1;
  else fail += 1;
  console.log(`${ok ? 'OK  ' : 'FAIL'} ${name}${ok || info === undefined ? '' : ' → ' + JSON.stringify(info).slice(0, 400)}`);
}

async function call(method: string, path: string, opts: { token?: string; tenant?: string; body?: unknown; ifMatch?: string } = {}) {
  const headers: Record<string, string> = { 'content-type': 'application/json' };
  if (opts.token) headers.authorization = `Bearer ${opts.token}`;
  if (opts.tenant) headers['x-tenant-id'] = opts.tenant;
  if (opts.ifMatch !== undefined) headers['if-match'] = opts.ifMatch;
  const res = await fetch(API + path, { method, headers, body: opts.body === undefined ? undefined : JSON.stringify(opts.body) });
  const text = await res.text();
  let body: any = text;
  try { body = JSON.parse(text); } catch { /* texto */ }
  return { status: res.status, body, requestId: res.headers.get('x-request-id') };
}

const suf = randomUUID().slice(0, 8);
const password = `S3cret-${randomUUID().slice(0, 6)}`;

async function org(letter: string, adminToken: string) {
  const email = `e2e-f1-${letter.toLowerCase()}-${suf}@example.test`;
  const reg = await call('POST', '/iam/auth/register-organization', {
    body: {
      organization: { code: `E2EF1_${letter}_${suf.toUpperCase()}`, legalName: `Hospital E2E ${letter} ${suf}`, tenantType: 'HOSPITAL', countryConceptId: CONCEPTS.COUNTRY_BO, jurisdictionConceptId: PROF.JURISDICTION_SEDES_SANTA_CRUZ },
      owner: { email, password, name: 'Dueña', lastName: letter },
    },
  });
  if (reg.status !== 201) throw new Error(`registro ${letter}: ${reg.status} ${JSON.stringify(reg.body)}`);
  const login = await call('POST', '/iam/auth/login', { body: { email, password } });
  const tenant = reg.body.tenantId as string;
  const token = login.body.accessToken as string;
  const practice = await call('POST', '/practices', { token: adminToken, body: { tenantId: tenant, code: `E2EF1-P-${letter}-${suf}`, name: `Hospital ${letter}`, timeZone: 'America/La_Paz' } });
  const site = await call('POST', `/practices/${practice.body.id}/sites`, { token, tenant, body: { code: `E2EF1-S-${letter}-${suf}`, name: `Sede ${letter}` } });
  if (site.status !== 201) throw new Error(`sede ${letter}: ${site.status} ${JSON.stringify(site.body)}`);
  return { tenant, token, siteId: site.body.id as string };
}

async function main() {
  const admin = await call('POST', '/iam/auth/login', { body: { email: env.BOOTSTRAP_ADMIN_EMAIL, password: env.BOOTSTRAP_ADMIN_PASSWORD } });
  if (admin.status !== 200) throw new Error(`login admin: ${admin.status}`);
  const A = await org('A', admin.body.accessToken);
  const B = await org('B', admin.body.accessToken);
  const asA = { token: A.token, tenant: A.tenant };
  const asB = { token: B.token, tenant: B.tenant };

  check('sede A con tenant_id de su práctica', sql(`select tenant_id from practice.practice_sites where id='${A.siteId}'`) === A.tenant);

  const cat = await call('GET', '/practice/infrastructure-catalog', asA);
  check('catálogo 200 con 10 formas y 30 funciones', cat.status === 200 && cat.body.forms.length === 10 && cat.body.functions.length === 30, cat.body);

  // --- alta
  const ed = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `ED-${suf}`, name: 'Edificio', formCode: 'LOC_FORM_BUILDING', functionCode: 'SPACE_FN_BUILDING_GENERIC', activate: true } });
  check('alta edificio 201 ACTIVO', ed.status === 201 && ed.body.status?.code === 'SPACE_ACTIVE', ed.body);
  const edRow = sql(`select tenant_id||'|'||form_concept_id||'|'||status_concept_id from practice.care_spaces where id='${ed.body.id}'`);
  check('fila: tenant, forma y estado', edRow === `${A.tenant}|${PRAC.LOC_FORM_BUILDING}|${PRAC.SPACE_ACTIVE}`, edRow);
  check('historia con tenant_id', sql(`select count(*) from audit.care_spaces_history where care_space_id='${ed.body.id}' and tenant_id='${A.tenant}'`) === '1');
  check('sello CARE_SPACE_CREATED', sql(`select count(*) from audit.audit_log where action='CARE_SPACE_CREATED' and entity_id='${ed.body.id}'`) === '1');
  check('evento practice.care_space.created', sql(`select count(*) from messaging.domain_events where event_type='practice.care_space.created' and aggregate_id='${ed.body.id}' and tenant_id='${A.tenant}'`) === '1');

  const bed = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `CAMA-RAIZ-${suf}`, name: 'Cama', formCode: 'LOC_FORM_BED', functionCode: 'SPACE_FN_BED_STANDARD' } });
  check('cama en la raíz 422 SPACE_PARENT_RULE (con requestId)', bed.status === 422 && bed.body.reason === 'SPACE_PARENT_RULE' && Boolean(bed.body.correlationId), bed.body);

  const room = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `301-${suf}`, name: 'Hab 301', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_INPATIENT_ROOM', parentSpaceId: ed.body.id, activate: true, areaM2: 18.5 } });
  check('local bajo edificio 201', room.status === 201, room.body);
  const bulk = await call('POST', `/sites/${A.siteId}/care-spaces/bulk`, { ...asA, body: { parentSpaceId: room.body.id, formCode: 'LOC_FORM_BED', functionCode: 'SPACE_FN_BED_STANDARD', codePattern: `301-${suf}-{A..C}`, namePattern: 'Cama {}', censable: true, activate: true } });
  check('alta masiva 201 ×3', bulk.status === 201 && bulk.body.created?.length === 3, bulk.body);
  check('3 camas en la base', sql(`select count(*) from practice.care_spaces where parent_space_id='${room.body.id}' and tenant_id='${A.tenant}'`) === '3');

  const dup = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `301-${suf}`, name: 'Dup', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_OFFICE_ADMIN' } });
  check('código repetido 409 SPACE_CODE_TAKEN', dup.status === 409 && dup.body.reason === 'SPACE_CODE_TAKEN', dup.body);

  // --- relectura
  const struct = await call('GET', `/sites/${A.siteId}/structure`, asA);
  check('structure 200 con 5 espacios y asOf', struct.status === 200 && struct.body.spaces.length === 5 && Boolean(struct.body.asOf), struct.body);
  const det = await call('GET', `/care-spaces/${room.body.id}`, asA);
  check('ficha: path, derivados (3 camas censables)', det.status === 200 && det.body.path[0]?.id === ed.body.id && det.body.derived?.censableBeds === 3, det.body.derived);

  // --- aislamiento
  const xs = [
    await call('GET', `/care-spaces/${room.body.id}`, asB),
    await call('GET', `/sites/${A.siteId}/structure`, asB),
    await call('PATCH', `/care-spaces/${room.body.id}`, { ...asB, ifMatch: '1', body: { name: 'x' } }),
    await call('POST', `/sites/${A.siteId}/care-spaces/bulk`, { ...asB, body: { formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_OFFICE_ADMIN', codePattern: 'X-{1..2}', namePattern: 'x' } }),
    await call('POST', `/care-spaces/${room.body.id}/lifecycle`, { ...asB, ifMatch: '1', body: { action: 'retire', reason: 'x' } }),
    await call('PATCH', `/sites/${A.siteId}/regulatory`, { ...asB, body: { snisCode: 'X' } }),
  ];
  check('desde B: todo 404', xs.every((r) => r.status === 404), xs.map((r) => r.status));
  check('nada cambió en A', sql(`select count(*) from practice.care_spaces where practice_site_id='${A.siteId}'`) === '5' && sql(`select name from practice.care_spaces where id='${room.body.id}'`) === 'Hab 301');

  // --- If-Match
  const noIf = await call('PATCH', `/care-spaces/${room.body.id}`, { ...asA, body: { name: 'Hab. 301' } });
  check('PATCH sin If-Match 428 IF_MATCH_REQUIRED', noIf.status === 428 && noIf.body.reason === 'IF_MATCH_REQUIRED', noIf.body);
  const bad = await call('PATCH', `/care-spaces/${room.body.id}`, { ...asA, ifMatch: '99', body: { name: 'Hab. 301' } });
  check('PATCH If-Match distinto 412 ROW_VERSION_MISMATCH', bad.status === 412 && bad.body.reason === 'ROW_VERSION_MISMATCH', bad.body);
  const rv = det.body.rowVersion;
  const ok = await call('PATCH', `/care-spaces/${room.body.id}`, { ...asA, ifMatch: `W/"${rv}"`, body: { name: 'Hab. 301', pressureCode: 'PRESSURE_NEGATIVE' } });
  check('PATCH 200 y rowVersion+1', ok.status === 200 && ok.body.rowVersion === rv + 1, ok.body);
  check('PATCH persistido', sql(`select name||'|'||pressure_concept_id||'|'||row_version from practice.care_spaces where id='${room.body.id}'`) === `Hab. 301|${PRAC.PRESSURE_NEGATIVE}|${rv + 1}`);
  const re = await call('GET', `/care-spaces/${room.body.id}`, asA);
  check('relectura muestra la presión', re.body.pressure?.code === 'PRESSURE_NEGATIVE', re.body.pressure);
  const imm = await call('PATCH', `/care-spaces/${room.body.id}`, { ...asA, ifMatch: String(rv + 1), body: { code: 'OTRO' } });
  check('código de activo 422 SPACE_IMMUTABLE_FIELD', imm.status === 422 && imm.body.reason === 'SPACE_IMMUTABLE_FIELD', imm.body);

  // --- mover
  const a1 = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `AR1-${suf}`, name: 'Área 1', formCode: 'LOC_FORM_AREA', functionCode: 'SPACE_FN_AREA_GENERIC' } });
  const a2 = await call('POST', `/sites/${A.siteId}/care-spaces`, { ...asA, body: { code: `AR2-${suf}`, name: 'Área 2', formCode: 'LOC_FORM_AREA', functionCode: 'SPACE_FN_AREA_GENERIC', parentSpaceId: a1.body.id } });
  const cyc = await call('POST', `/care-spaces/${a1.body.id}/move`, { ...asA, ifMatch: String(a1.body.rowVersion), body: { parentSpaceId: a2.body.id } });
  check('mover padre bajo su hijo 422 SPACE_CYCLE', cyc.status === 422 && cyc.body.reason === 'SPACE_CYCLE', cyc.body);
  const mv = await call('POST', `/care-spaces/${a2.body.id}/move`, { ...asA, ifMatch: String(a2.body.rowVersion), body: { parentSpaceId: null } });
  check('mover a la raíz 200 y persistido', mv.status === 200 && sql(`select coalesce(parent_space_id::text,'null') from practice.care_spaces where id='${a2.body.id}'`) === 'null', mv.body);
  check('evento moved', sql(`select count(*) from messaging.domain_events where event_type='practice.care_space.moved' and aggregate_id='${a2.body.id}'`) === '1');

  // --- retiro
  const rc = await call('GET', `/care-spaces/${room.body.id}/retire-check`, asA);
  check('retire-check: ACTIVE_CHILDREN 3 + no verificables', rc.status === 200 && rc.body.blockers[0]?.kind === 'ACTIVE_CHILDREN' && rc.body.blockers[0]?.count === 3 && rc.body.unverifiable.length === 2, rc.body);
  const blk = await call('POST', `/care-spaces/${room.body.id}/lifecycle`, { ...asA, ifMatch: String(rv + 1), body: { action: 'retire', reason: 'remodelación' } });
  check('retirar con hijos 409 SPACE_RETIRE_BLOCKED con próxima acción', blk.status === 409 && blk.body.reason === 'SPACE_RETIRE_BLOCKED' && Boolean(blk.body.details?.blockers?.[0]?.nextAction?.label), blk.body);
  const bedC = bulk.body.created[2];
  const ret = await call('POST', `/care-spaces/${bedC.id}/lifecycle`, { ...asA, ifMatch: String(bedC.rowVersion), body: { action: 'retire', reason: 'baja de cama' } });
  check('retirar hoja 200 RETIRED', ret.status === 200 && ret.body.status?.code === 'SPACE_RETIRED', ret.body);
  check('retiro persistido con valid_to', sql(`select (status_concept_id='${PRAC.SPACE_RETIRED}' and valid_to is not null)::text from practice.care_spaces where id='${bedC.id}'`) === 'true');
  check('motivo en la historia, no en el evento', sql(`select data_snapshot->>'lifecycleReason' from audit.care_spaces_history where care_space_id='${bedC.id}' order by revision_no desc limit 1`) === 'baja de cama' && sql(`select count(*) from messaging.domain_events where aggregate_id='${bedC.id}' and payload_json::text like '%baja de cama%'`) === '0');
  const hist = await call('GET', `/care-spaces/${bedC.id}/history`, asA);
  check('historial: retiro primero, con motivo y actor; una sola alta', hist.status === 200 && hist.body[0]?.action === 'retire' && hist.body[0]?.reason === 'baja de cama' && Boolean(hist.body[0]?.actorName) && hist.body.filter((h: any) => h.action === 'created').length === 1, hist.body);
  check('historial de A desde B: 404', (await call('GET', `/care-spaces/${bedC.id}/history`, asB)).status === 404);
  const re2 = await call('GET', `/care-spaces/${room.body.id}`, asA);
  check('relectura: 2 camas', re2.body.derived?.beds === 2, re2.body.derived);

  // --- importación
  const imp = await call('POST', `/sites/${A.siteId}/care-spaces/import?dryRun=true`, { ...asA, body: { rows: [
    { code: `IMP1-${suf}`, name: 'Oficina', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_OFFICE_ADMIN', parentCode: `ED-${suf}` },
    { code: `IMP2-${suf}`, name: 'Cama', formCode: 'LOC_FORM_BED', functionCode: 'SPACE_FN_BED_STANDARD' },
  ] } });
  check('import dryRun: ok false, errores por línea, nada escrito', imp.status === 200 && imp.body.ok === false && imp.body.rows[0].status === 'ok' && imp.body.rows[1].status === 'error' && Boolean(imp.body.rows[1].errors[0]?.message) && sql(`select count(*) from practice.care_spaces where code like 'IMP%-${suf}'`) === '0', imp.body);
  const imp2 = await call('POST', `/sites/${A.siteId}/care-spaces/import?dryRun=false`, { ...asA, body: { rows: [imp.body && { code: `IMP1-${suf}`, name: 'Oficina', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_OFFICE_ADMIN', parentCode: `ED-${suf}` }] } });
  check('import real: created 1 y persistido', imp2.status === 200 && imp2.body.created === 1 && sql(`select count(*) from practice.care_spaces where code='IMP1-${suf}'`) === '1', imp2.body);

  // --- unidades y vínculos
  const u = await call('POST', `/sites/${A.siteId}/clinical-units`, { ...asA, body: { code: `UNI-${suf}`, name: 'Internación' } });
  check('unidad 201 con tenant', u.status === 201 && sql(`select tenant_id from practice.clinical_units where id='${u.body.id}'`) === A.tenant, u.body);
  const as1 = await call('POST', `/care-spaces/${room.body.id}/unit-assignments`, { ...asA, body: { clinicalUnitId: u.body.id, roleCode: 'SPACE_UNIT_ROLE_OWNER' } });
  check('vínculo OWNER 201', as1.status === 201, as1.body);
  const u2 = await call('POST', `/sites/${A.siteId}/clinical-units`, { ...asA, body: { code: `UNI2-${suf}`, name: 'Otra' } });
  const as2 = await call('POST', `/care-spaces/${room.body.id}/unit-assignments`, { ...asA, body: { clinicalUnitId: u2.body.id, roleCode: 'SPACE_UNIT_ROLE_OWNER' } });
  check('segundo OWNER 409', as2.status === 409 && as2.body.reason === 'SPACE_OWNER_TAKEN', as2.body);
  const ud = await call('GET', `/clinical-units/${u.body.id}`, asA);
  check('ficha de unidad con el vínculo', ud.status === 200 && ud.body.assignments.length === 1, ud.body);
  const ur = await call('POST', `/clinical-units/${u.body.id}/lifecycle`, { ...asA, ifMatch: String(ud.body.rowVersion), body: { action: 'retire', reason: 'x' } });
  check('retirar unidad con espacio vinculado 409 UNIT_RETIRE_BLOCKED', ur.status === 409 && ur.body.reason === 'UNIT_RETIRE_BLOCKED', ur.body);

  // --- establecimiento
  const tc = await call('PATCH', `/sites/${A.siteId}/regulatory`, { ...asA, body: { typologyCode: 'TYPOLOGY_HEALTH_POST' } });
  check('tipología Puesto de Salud con camas 409 TYPOLOGY_CONFLICT', tc.status === 409 && tc.body.reason === 'TYPOLOGY_CONFLICT' && tc.body.details.conflicts.length >= 1 && Boolean(tc.body.details.conflicts[0].name) && Boolean(tc.body.details.conflicts[0].function?.code), tc.body);
  const rg = await call('PATCH', `/sites/${A.siteId}/regulatory`, { ...asA, body: { typologyCode: 'TYPOLOGY_HOSPITAL_L2', snisCode: `S${suf}`, sedesCode: 'SEDES_SCZ', careLevelCode: 'CARE_LEVEL_2' } });
  check('regulatorio 200 y faltantes obligatorios', rg.status === 200 && rg.body.missingRequiredFunctions.includes('SPACE_FN_OPERATING_ROOM'), rg.body);
  check('regulatorio persistido', sql(`select facility_typology_concept_id||'|'||snis_code from practice.practice_sites where id='${A.siteId}'`) === `${PRAC.TYPOLOGY_HOSPITAL_L2}|S${suf}`);
  const vsOwner = await call('POST', `/sites/${A.siteId}/regulatory/verify-snis`, asA);
  check('verificar SNIS como owner 403', vsOwner.status === 403, vsOwner.body);
  const vs = await call('POST', `/sites/${A.siteId}/regulatory/verify-snis`, { token: admin.body.accessToken, tenant: A.tenant });
  check('verificar SNIS como plataforma 200', vs.status === 200 && vs.body.regulatory?.snisVerified === true, vs.body);
  check('snis_verified_at en la base', sql(`select (snis_verified_at is not null)::text from practice.practice_sites where id='${A.siteId}'`) === 'true');
  await call('PATCH', `/sites/${B.siteId}/regulatory`, { ...asB, body: { snisCode: `S${suf}` } });
  const vsB = await call('POST', `/sites/${B.siteId}/regulatory/verify-snis`, { token: admin.body.accessToken, tenant: B.tenant });
  check('verificar en B el código ya verificado en A: 409 genérico', vsB.status === 409 && vsB.body.reason === 'SNIS_CODE_UNAVAILABLE' && !JSON.stringify(vsB.body).includes(A.siteId), vsB.body);

  const tb = await call('PATCH', `/sites/${B.siteId}/regulatory`, { ...asB, body: { typologyCode: 'TYPOLOGY_HEALTH_POST' } });
  const orB = await call('POST', `/sites/${B.siteId}/care-spaces`, { ...asB, body: { code: `QX-${suf}`, name: 'Quirófano', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_OPERATING_ROOM' } });
  check('Puesto de Salud: quirófano 422 TYPOLOGY_NOT_ALLOWED; consultorio 201', tb.status === 200 && orB.status === 422 && orB.body.reason === 'TYPOLOGY_NOT_ALLOWED' && (await call('POST', `/sites/${B.siteId}/care-spaces`, { ...asB, body: { code: `CONS-${suf}`, name: 'Consultorio', formCode: 'LOC_FORM_ROOM', functionCode: 'SPACE_FN_CONSULTING_ROOM' } })).status === 201, orB.body);
  const noIfMatch428 = await call('POST', `/care-spaces/${room.body.id}/move`, { ...asA, body: { parentSpaceId: null } });
  check('move sin If-Match: 428', noIfMatch428.status === 428 && noIfMatch428.body.reason === 'IF_MATCH_REQUIRED', noIfMatch428.body);
  const lic = await call('POST', `/sites/${A.siteId}/licenses`, { ...asA, body: { authorityName: 'SEDES Santa Cruz', resolutionNumber: `RA-${suf}`, issuedOn: '2020-01-01', expiresOn: '2021-01-01', notes: '' } });
  check('habilitación 201 en trámite', lic.status === 201 && lic.body.status.code === 'LIC_IN_PROCESS', lic.body);
  check('habilitación en facility_licenses', sql(`select practice_site_id from organization_extensions.facility_licenses where id='${lic.body.id}'`) === A.siteId);
  const lv = await call('PATCH', `/site-licenses/${lic.body.id}`, { token: admin.body.accessToken, tenant: A.tenant, body: { status: 'LIC_VALID' } });
  check('plataforma la verifica; vencida se lee LIC_EXPIRED', lv.status === 200 && lv.body.status.code === 'LIC_EXPIRED', lv.body);
  const dc = await call('POST', `/sites/${A.siteId}/declared-capacities`, { ...asA, body: { kindCode: 'CAP_BEDS_CENSABLE', quantity: 10, asOf: '2026-01-01' } });
  check('capacidad declarada vs derivada: brecha 10 vs 2', dc.status === 201 && JSON.stringify(dc.body.gaps).includes('"declared":10,"derived":2'), dc.body);

  console.log(`\n${pass} OK, ${fail} FAIL`);
  process.exitCode = fail ? 1 : 0;
}

main().catch((e) => {
  console.error('ERROR', e.message);
  process.exitCode = 2;
});
