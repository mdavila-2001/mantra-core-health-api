import { MikroORM } from '/home/user/mantra-core-health-api/node_modules/@mikro-orm/postgresql';
import { PractitionerContextAdapter } from '/home/user/mantra-core-health-api/src/modules/profiles/infrastructure/adapters/practitioner-context.adapter';
import { PROF } from '/home/user/mantra-core-health-api/src/modules/profiles/profiles.concepts';
import { COMM } from '/home/user/mantra-core-health-api/src/modules/community/community.concepts';
import { CONCEPTS } from '/home/user/mantra-core-health-api/src/common';

async function main() {
const orm = await MikroORM.init({
  host: '/tmp', port: 55432, user: 'postgres', dbName: 'postgres',
  entities: [], discovery: { warnWhenNoEntities: false }, allowGlobalContext: true,
});
const em = orm.em.fork();
const x = (sql: string, p: unknown[] = []) => em.getConnection().execute(sql, p, 'run');
const DR_A = '11111111-1111-1111-1111-111111111111', U_A = 'aaaaaaaa-0000-0000-0000-000000000001';
const DR_B = '22222222-2222-2222-2222-222222222222', U_B = 'aaaaaaaa-0000-0000-0000-000000000002';
const DR_C = '33333333-3333-3333-3333-333333333333'; // sin cuenta, con reseñas en perfil OCULTO
const OTHER = '99999999-9999-9999-9999-999999999999';
await x(`INSERT INTO profiles.person_account_links(person_id,user_id,status_concept_id) VALUES (?,?,?),(?,?,?),(?,?,?)`,
  [DR_A, U_A, PROF.ACCOUNT_LINK_ACTIVE, DR_B, U_B, PROF.ACCOUNT_LINK_ACTIVE, DR_B, 'aaaaaaaa-0000-0000-0000-000000000003', OTHER]);
for (let i = 0; i < 5; i++) await x(`INSERT INTO clinical.encounters(created_by_user_id) VALUES (?)`, [U_A]);
for (let i = 0; i < 2; i++) await x(`INSERT INTO clinical.encounters(created_by_user_id) VALUES (?)`, [U_B]);
await x(`INSERT INTO clinical.encounters(created_by_user_id) VALUES ('aaaaaaaa-0000-0000-0000-000000000003')`);
for (let i = 0; i < 3; i++) await x(`INSERT INTO clinical.medication_requests(created_by_user_id) VALUES (?)`, [U_A]);
await x(`INSERT INTO chart.clinical_note_headers(created_by_user_id) VALUES (?)`, [U_B]);
await x(`INSERT INTO chart.document_records(created_by_user_id) VALUES (?)`, [U_A]);
const pub = (target: string, vis: string) => em.getConnection().execute<{id:string}[]>(
  `INSERT INTO community.public_profiles(target_type_concept_id,target_id,visibility_concept_id,status_concept_id) VALUES (?,?,?,?) RETURNING id`,
  [COMM.PROFILE_TARGET_PRACTITIONER, target, vis, CONCEPTS.STATE_ACTIVE], 'all');
const [pa] = await pub(DR_A, COMM.PROFILE_VISIBILITY_PUBLIC);
const [pc] = await pub(DR_C, OTHER);
const rev = (pp: string, r: number, mod = OTHER) => x(
  `INSERT INTO community.service_reviews(target_public_profile_id,overall_rating,publication_status_concept_id,moderation_status_concept_id) VALUES (?,?,?,?)`,
  [pp, r, COMM.PUBLICATION_PUBLISHED, mod]);
await rev(pa.id, 5); await rev(pa.id, 4); await rev(pa.id, 4);
await rev(pa.id, 1, COMM.MODERATION_REMOVED); // retirada: no cuenta
await rev(pc.id, 5); // perfil oculto: no cuenta

const adapter = new PractitionerContextAdapter({} as any);
const stats = await adapter.directoryStats(em, [DR_A, DR_B, DR_C]);
console.log(JSON.stringify(Object.fromEntries(stats), null, 1));
console.log('empty:', (await adapter.directoryStats(em, [])).size);
await orm.close();
}
main().catch((e) => { console.error(e); process.exit(1); });
